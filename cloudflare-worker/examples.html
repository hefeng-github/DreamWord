JavaScript/TypeScript

GET /api/lookup?word=hello
async function lookupWord(word) {
  const response = await fetch(
    'https://your-worker.workers.dev/api/lookup?word=' + encodeURIComponent(word)
  );
  const data = await response.json();
  return data;
}

// 使用示例
const result = await lookupWord('hello');
console.log(result.definitions);

Python

GET /api/lookup?word=hello
import requests

def lookup_word(word):
  url = f"https://your-worker.workers.dev/api/lookup?word={word}"
  response = requests.get(url)
  return response.json()

# 使用示例
result = lookup_word('hello')
print(result['definitions'])
